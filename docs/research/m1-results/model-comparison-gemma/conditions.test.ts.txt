import { describe, it, expect, beforeEach, afterEach } from 'vitest';
import { execFileSync } from 'node:child_process';
import { mkdtempSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { scan } from './conditions';
import type { GameRecord } from '../src/sim/game';

const mkRec = (
  gameId: number,
  startFen: string,
  start: [Record<string, number>, Record<string, number>],
  moves: [Record<string, number>, Record<string, number>],
  survived: [Record<string, number>, Record<string, number>],
  captures: [Record<string, number>, Record<string, number>],
  result: 1 | 0.5 | 0,
  reason: GameRecord['reason'],
  plies: number,
): GameRecord => ({
  gameId,
  pairId: gameId,
  colourSwapped: false,
  configId: 'test',
  seed: gameId,
  backRankWhite: startFen.split(' ')[0].split('/')[7],
  backRankBlack: startFen.split('/')[0],
  openingPlies: 0,
  startFen,
  result,
  reason,
  plies,
  ms: 0,
  firstCapturePly: null,
  coverage: 0,
  moves: [],
  events: {},
  stats: [
     { start: start[0], moves: moves[0], survived: survived[0], captures: captures[0], taken: {} },
     { start: start[1], moves: moves[1], survived: survived[1], captures: captures[1], taken: {} },
  ],
} as unknown as GameRecord);

let dir = '';
const writeRun = (id: string, recs: GameRecord[]): string => {
  const p = join(dir, `${id}.jsonl`);
  writeFileSync(p, recs.map(r => JSON.stringify(r)).join('\n') + '\n');
  return p;
};

describe('conditions scan', () => {
  beforeEach(() => { dir = mkdtempSync(join(tmpdir(), 'cond-')); });
  afterEach(() => { rmSync(dir, { recursive: true, force: true }); });

  it('counts owner armies by type, presence, moves, captures and end presence', async () => {
    // Synthetic fixtures, not played games. Two Archers on one side are TWO PIECES IN ONE ARMY,
    // never two armies: each owning side adds exactly one owner.
    // Game 1: White start A=2 (one army), moves A=4, survived A=2, captures A=1;
    // Black start A=1, moves A=0 (a zero-move owner), survived A=0, captures A=0.
    const r1 = mkRec(1, 'a6k/8/8/8/8/8/8/A5AK w - - 0 1',
       [{ A: 2, K: 1 }, { A: 1, K: 1 }],
       [{ A: 4 }, { A: 0 }],
       [{ A: 2, K: 1 }, { A: 0, K: 1 }],
       [{ A: 1 }, { A: 0 }],
       1, 'checkmate', 10);
    // Game 2: White has no A; Black start A=1, moves A=2, survived A=1, captures A=1.
    // White also owns one Guard (G=1, no move, survived) so Guard owners=1 and present=1.
    const r2 = mkRec(2, 'a6k/8/8/8/8/8/2G5/7K w - - 0 1',
       [{ G: 1, K: 1 }, { A: 1, K: 1 }],
       [{ G: 0 }, { A: 2 }],
       [{ G: 1, K: 1 }, { A: 1, K: 1 }],
       [{ G: 0 }, { A: 1 }],
       0.5, 'draw50', 12);
    const file = writeRun('archers', [r1, r2]);

    const st = await scan(file);
    expect(st.games).toBe(2);
    const a = st.pieces.get('A');
    expect(a).toBeDefined();
    expect(a!.owners).toBe(3);
    expect(a!.present).toBe(2);
    expect(a!.moves).toBe(6);
    expect(a!.zero).toBe(1);
    expect(a!.survived).toBe(2);
    expect(a!.captures).toBe(2);
    const g = st.pieces.get('G');
    expect(g).toBeDefined();
    expect(g!.owners).toBe(1);
    expect(g!.present).toBe(1);
   });

  it('ignores invalid piece keys and nonpositive starting counts', async () => {
    const r1 = mkRec(1, 'a6k/8/8/8/8/8/8/A5AK w - - 0 1',
       [{ A: 1, a: 5, '3': 2, '': 2, ' ': 2, PN: 2, G: 0, B: -1 }, { A: 1 }],
       [{ A: 1 }, { A: 1 }],
       [{ A: 1 }, { A: 1 }],
       [{ A: 0 }, { A: 0 }],
       1, 'checkmate', 5);
    const file = writeRun('mixed', [r1]);

    const st = await scan(file);
    expect(st.pieces.get('a')).toBeUndefined();
    expect(st.pieces.get('3')).toBeUndefined();
    expect([...st.pieces.keys()]).toEqual(['A']);
    expect(st.pieces.get('A')!.owners).toBe(2);
   });

  it('returns empty aggregates for an empty run', async () => {
    const file = writeRun('empty', []);
    const st = await scan(file);
    expect(st.games).toBe(0);
    expect(st.pieces.size).toBe(0);
   });
});

describe('conditions CLI', () => {
  beforeEach(() => { dir = mkdtempSync(join(tmpdir(), 'cond-cli-')); });
  afterEach(() => { rmSync(dir, { recursive: true, force: true }); });

  it('prints owner armies, presence, moves per owner and still share from synthetic records', () => {
    // Same synthetic fixture as the scan test, written to a temp file the CLI reads.
    const r1 = mkRec(1, 'a6k/8/8/8/8/8/8/A5AK w - - 0 1',
       [{ A: 2, K: 1 }, { A: 1, K: 1 }],
       [{ A: 4 }, { A: 0 }],
       [{ A: 2, K: 1 }, { A: 0, K: 1 }],
       [{ A: 1 }, { A: 0 }],
       1, 'checkmate', 10);
    const r2 = mkRec(2, 'a6k/8/8/8/8/8/2G5/7K w - - 0 1',
       [{ G: 1, K: 1 }, { A: 1, K: 1 }],
       [{ G: 0 }, { A: 2 }],
       [{ G: 1, K: 1 }, { A: 1, K: 1 }],
       [{ G: 0 }, { A: 1 }],
       0.5, 'draw50', 12);
    const file = writeRun('cli', [r1, r2]);

    const out = execFileSync('node_modules/.bin/tsx',
       ['tools/conditions.ts', '--runs', file],
       { stdio: 'pipe' }).toString();

    const rows = out.split('\n').map(line => line.split('|').map(cell => cell.trim()));
    expect(rows.find(row => row[0] === 'A')).toEqual(['A', '2/2 (100.0%)', '3', '2.00', '33.3%', '66.7%', '0.67']);
    expect(rows.find(row => row[0] === 'G')).toEqual(['G', '1/2 (50.0%)', '1', '0.00', '100.0%', '100.0%', '0.00']);
    expect(out).toContain('TYPE PRESENT AT END');
    expect(out).toContain('promotions');
    expect(out).not.toContain('NaN');
    expect(out).not.toContain('Infinity');
   });

  it('prints owner armies, presence, moves per owner and still share from synthetic records', () => {
    // Same synthetic fixture as the scan test, written to a temp file the CLI reads.
    const r1 = mkRec(1, 'a6k/8/8/8/8/8/8/A5AK w - - 0 1',
       [{ A: 2, K: 1 }, { A: 1, K: 1 }],
       [{ A: 4 }, { A: 0 }],
       [{ A: 2, K: 1 }, { A: 0, K: 1 }],
       [{ A: 1 }, { A: 0 }],
       1, 'checkmate', 10);
    const r2 = mkRec(2, 'a6k/8/8/8/8/8/2G5/7K w - - 0 1',
       [{ G: 1, K: 1 }, { A: 1, K: 1 }],
       [{ G: 0 }, { A: 2 }],
       [{ G: 1, K: 1 }, { A: 1, K: 1 }],
       [{ G: 0 }, { A: 1 }],
       0.5, 'draw50', 12);
    const file = writeRun('cli', [r1, r2]);

    const out = execFileSync('node_modules/.bin/tsx',
       ['tools/conditions.ts', '--runs', file],
       { stdio: 'pipe' }).toString();

    const rows = out.split('\n').map(line => line.split('|').map(cell => cell.trim()));
    expect(rows.find(row => row[0] === 'A')).toEqual(['A', '2/2 (100.0%)', '3', '2.00', '33.3%', '66.7%', '0.67']);
    expect(rows.find(row => row[0] === 'G')).toEqual(['G', '1/2 (50.0%)', '1', '0.00', '100.0%', '100.0%', '0.00']);
    expect(out).toContain('TYPE PRESENT AT END');
    expect(out).toContain('promotions');
    expect(out).not.toContain('NaN');
    expect(out).not.toContain('Infinity');
   });

  it('handles an empty run without NaN or Infinity', () => {
    const file = writeRun('empty', []);

    const out = execFileSync('node_modules/.bin/tsx',
       ['tools/conditions.ts', '--runs', file],
       { stdio: 'pipe' }).toString();

    expect(out).not.toContain('NaN');
    expect(out).not.toContain('Infinity');
  });

  describe('paired validation', () => {
    it('calculates mean for matching IDs', () => {
      const rA = mkRec(10, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const rB = mkRec(10, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 0.5, 'checkmate', 10);
      const fA = writeRun('pairA', [rA]);
      const fB = writeRun('pairB', [rB]);

      const out = execFileSync('node_modules/.bin/tsx',
         ['tools/conditions.ts', '--pair', `${fA},${fB}`],
         { stdio: 'pipe' }).toString();

      expect(out).toContain('result +0.500');
    });

    it('errors on duplicate gameId in a file', () => {
      const r1 = mkRec(1, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const r2 = mkRec(1, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const f = writeRun('dup', [r1, r2]);

      expect(() => {
        execFileSync('node_modules/.bin/tsx', ['tools/conditions.ts', '--pair', `${f},${f}`], { stdio: 'pipe' });
      }).toThrow();
    });

    it.each([
      ['startFen', (r: any) => ({ ...r, startFen: 'diff' })],
      ['seed', (r: any) => ({ ...r, seed: 999 })],
      ['openingPlies', (r: any) => ({ ...r, openingPlies: 5 })],
      ['colourSwapped', (r: any) => ({ ...r, colourSwapped: true })],
    ])('errors on %s mismatch', (field, mutate) => {
      const rA = mkRec(10, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const rB = mutate(mkRec(10, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 0, 'checkmate', 10));
      const fA = writeRun(`pairA_${field}`, [rA]);
      const fB = writeRun(`pairB_${field}`, [rB]);

      expect(() => {
        execFileSync('node_modules/.bin/tsx', ['tools/conditions.ts', '--pair', `${fA},${fB}`], { stdio: 'pipe' });
      }).toThrow();
    });

    it('prints unmatched counts', () => {
      const rA = mkRec(1, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const rB = mkRec(2, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 0, 'checkmate', 10);
      const fA = writeRun('unmatchA', [rA]);
      const fB = writeRun('unmatchB', [rB]);

      const out = execFileSync('node_modules/.bin/tsx',
         ['tools/conditions.ts', '--pair', `${fA},${fB}`],
         { stdio: 'pipe' }).toString();

      expect(out).toContain('paired 0 games');
    });

    it('handles zero pairs: no estimate available', () => {
      const rA = mkRec(1, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const rB = mkRec(2, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 0, 'checkmate', 10);
      const fA = writeRun('zeroA', [rA]);
      const fB = writeRun('zeroB', [rB]);

      const out = execFileSync('node_modules/.bin/tsx',
         ['tools/conditions.ts', '--pair', `${fA},${fB}`],
         { stdio: 'pipe' }).toString();

      expect(out).not.toContain('±');
    });

    it('handles one pair: mean only, no interval', () => {
      const rA = mkRec(1, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 1, 'checkmate', 10);
      const rB = mkRec(1, 'fen', [{K:1}, {K:1}], [{}, {}], [{}, {}], [{}, {}], 0, 'checkmate', 10);
      const fA = writeRun('oneA', [rA]);
      const fB = writeRun('oneB', [rB]);

      const out = execFileSync('node_modules/.bin/tsx',
         ['tools/conditions.ts', '--pair', `${fA},${fB}`],
         { stdio: 'pipe' }).toString();

      expect(out).toContain('result +1.000');
      expect(out).not.toContain('±');
    });
  });

});
